% Use \Count for intrinsically integral quantities and \Raw for continuous
% quantities. \Percent is useful outside S columns; percentage columns in the
% metric tables apply the same three-significant-figure policy automatically.
\newcommand{\Count}[1]{\num[round-mode=none]{#1}}
% `exponent-thresholds` keeps the middle of the range in plain decimal and
% sends only the extremes to scientific notation, so `lr_g` reads
% 3.000x10^{-5} instead of 0.00003000 while a 58.07-second phase stays 58.07.
\newcommand{\Raw}[1]{\num[
  round-mode        = figures,
  round-precision   = 4,
  exponent-mode     = threshold,
  exponent-thresholds = -3:5
]{#1}}
\newcommand{\Percent}[1]{\num[round-mode=places,round-precision=1]{#1}\,\%}

% Scaled by 10^3 in the generator, so a real run spans 0.08472..303.0 across
% the three metrics: three integer digits, five decimals at four significant
% figures.
\newcolumntype{D}{S[
  round-mode=figures,
  round-precision=4,
  table-format=3.5
]}
\newcolumntype{P}{S[
  round-mode=places,
  round-precision=1,
  table-format=+4.1
]}

% still what the single-page diagnostics use; this one adds `page=` and drops
% the height cap, because a level figure page is square and fits the text
% width -- capping its height is what made the panels unreadable.
\newcommand{\ReportPage}[2]{\IfFileExists{#1}{
    % A six-panel page is square, so it is width-limited and leaves roughly a
    % third of the page height spare; the fills centre it rather than letting
    % it ride the top margin.

    \includegraphics[width=\linewidth,page=#2]{#1}
    \end{center}
    \vspace*{\fill}
}{
    \fbox{\parbox[c][0.4\textheight][c]{0.94\linewidth}{
        \centering\color{ReportGray}
        Missing figure\par\medskip
        \ttfamily\detokenize{#1}}}
  }
